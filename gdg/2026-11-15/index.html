たのみ方

やりたいことを、日本語でそのまま書くだけです。

  • ひとつだけたのむ(まとめて頼むと失敗します)
  • 最後に「変更したjavascript部分のコード全体を出してください」とつける
書き方の見本〔 〕のところに、やりたいことを1つ書く

やってみたいこと

  • 音をもっと速く／ゆっくり
  • 音を増やす・低い音を足す
  • 色を変える・虹色にする
  • 残像が残るようにする
  • 四角や三角もまぜる
  • 4回に1回、強い音を入れる
  • おさえたまま動かすと鳴るようにする
  • キーボードでも鳴るようにする

文は自分で考えます。思いついたものを作ってください。

うまくいかないとき

赤いエラーが出た「次のエラーが出ました。原因を直したjavascript部分のコード全体を出してください。」
音が出ない・画面が真っ黒「実行しても音が出ません。原因を調べて、動くjavascript部分のコード全体を出してください。」
音が1オクターブ低い「音名ではなく、MIDIノート番号と midiToFreq() を使ってください。」
前より悪くなった「さっきの状態に戻してください。」
一部しか返ってこない「javascript部分のコード全体を出してください。」

どうにもならないときは手を挙げてください。それが一番早いです。

ゼロから作る

p5.js Web Editor で新しいスケッチを作り、3つとも全部を貼りかえます。

p5.js Web Editor をひらく

sketch.js1番目に貼る
sketch.js の中身を見る
// 音と光 / p5.js
//
// --- 変更履歴 ------------------------------------
// v1  一定のリズムで音が鳴り、鳴るたびに円が広がる
// --- ここまで ------------------------------------
// ※ 変更したら「ここまで」の行の1つ上に、v2, v3 … と1行だけ足してください。
//   すでにある行は書き換えないでください。

// ---- この3つを変えるだけでも、かなり変わります ----
const ONKAI  = [60, 62, 64, 67, 69];   // 使う音(ド レ ミ ソ ラ)
const SOKUDO = 2000;  // 自動で鳴る音の間隔(ミリ秒)。小さいほど速い
const IRO    = 200;   // 色あい(0〜360)
// ---------------------------------------------------
// ※ 音は数字で指定します。1 増やすと半音上、12 増やすと1オクターブ上。
//    'A4' のような音名は使わないでください。p5.sound の不具合で、
//    A と B だけ1オクターブ低く鳴ります。

let synth;            // 音を出すもの
let hajimatta = false;
let tsugi = 0;        // 次に鳴らす時刻
let marus = [];       // 画面に出ている円

function setup() {
  createCanvas(800, 500);
  colorMode(HSB, 360, 100, 100, 100);
  synth = new p5.PolySynth();
  noStroke();
}

function draw() {
  background(0, 0, 8);

  // まだ始まっていないときは案内を出す
  if (!hajimatta) {
    fill(0, 0, 100);
    textAlign(CENTER, CENTER);
    textSize(24);
    text('画面をクリックすると始まります', width / 2, height / 2);
    return;
  }

  // 時間がきたら、自動で音をひとつ鳴らす
  if (millis() > tsugi) {
    narasu(floor(random(ONKAI.length)), random(width), random(height), false);
    tsugi = millis() + SOKUDO;
  }

  // 円を描く(だんだん大きく、だんだん うすく)
  for (let i = marus.length - 1; i >= 0; i--) {
    const m = marus[i];
    m.ookisa += 3;    // だんだん大きく
    m.nokori -= 1;    // だんだん うすく
    if (m.nokori <= 0) {
      marus.splice(i, 1);
      continue;
    }
    fill(m.iro, 70, 100, m.nokori * m.koi);   // koi = こさ
    circle(m.x, m.y, m.ookisa);
  }
}

// 音をひとつ鳴らして、同じ場所に円をひとつ出す
// jibun が true のときは、自分でクリックした音。大きく・長く鳴らす
function narasu(bangou, x, y, jibun) {
  // 自分で鳴らした音は、大きく・長く・こく。自動の音はひかえめに
  const tsuyosa = jibun ? 0.55 : 0.12;   // 音の大きさ
  const nagasa  = jibun ? 1.2  : 0.4;    // 音の長さ(秒)
  const koi     = jibun ? 1.0  : 0.35;   // 円のこさ(1でそのまま、小さいほど うすい)

  synth.play(midiToFreq(ONKAI[bangou]), tsuyosa, 0, nagasa);
  marus.push({
    x: x,
    y: y,
    ookisa: jibun ? 55 : 20,   // はじめの大きさ
    nokori: 100,               // 消えるまでのカウント(こさとは別)
    koi: koi,
    iro: (IRO + bangou * 70) % 360
  });
}

// 画面をクリックしたとき
// (ブラウザの決まりで、一度クリックしないと音を出せません)
function mousePressed() {
  if (!hajimatta) {
    userStartAudio();
    hajimatta = true;
    return;
  }
  // クリックした高さで、音の高さが変わる
  const bangou = floor(map(mouseY, height, 0, 0, ONKAI.length));
  narasu(constrain(bangou, 0, ONKAI.length - 1), mouseX, mouseY, true);
}
