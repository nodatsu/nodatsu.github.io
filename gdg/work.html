// 音に反応する四角 / p5.js
//
// [変更履歴] 変更するたびに、いちばん下に1行だけ追記する。
//            前の行は書き換えない。
// v1  白い四角が、音の大きさに合わせて伸び縮みする

const OTO_URL = 'dojo_loop.mp3';   // 音源の場所

let song;      // 音楽
let amp;       // 音の大きさをはかるもの
let playing = false;

function preload() {
  song = loadSound(OTO_URL);
}

function setup() {
  createCanvas(960, 540);
  amp = new p5.Amplitude();
  rectMode(CENTER);          // 四角を「中心の位置」で置く
  noStroke();
}

function draw() {
  background(0);             // 背景を黒でぬりつぶす

  // まだ音が鳴っていないときは案内を出す
  if (!playing) {
    fill(255);
    textAlign(CENTER, CENTER);
    textSize(26);
    text('画面をクリックすると音が鳴ります', width / 2, height / 2);
    return;
  }

  const oto = amp.getLevel();        // 今の音の大きさ(0〜0.3くらい)
  const ookisa = 60 + oto * 1000;    // 音が大きいほど大きく

  fill(255);
  rect(width / 2, height / 2, ookisa, ookisa);
}

// 画面をクリックしたら音を鳴らす
// (ブラウザの決まりで、一度クリックしないと音を出せません)
function mousePressed() {
  if (!playing) {
    userStartAudio();
    song.loop();
    playing = true;
  }
}

3改造例

ここから先は、自分の言葉でAIに頼みます。やりたいことを日本語で書くだけです。

書き方の見本 ・「〜してください」と書く
・ひとつだけ書く(まとめて頼むと失敗します)
・最後に「変更したコード全体を出してください」とつけると確実
例: 色を変える図形の色を、明るい暖色系に変えてください。

4うまくいかないとき

ひとつずつ頼むまとめて頼むと失敗します
動いたら保存コードをどこかにコピー
エラーはAIに貼る英語は読まなくていい

下の文はそのまま使ってください。ここは考えなくて大丈夫です。

赤いエラーが出た「次のエラーが出ました。原因を直したコード全体を出してください。」
↑これを送って、そのあとにエラーを全部貼る
画面が真っ白「実行しても画面が真っ白のままです。原因を調べて、動くコード全体を出してください。」
前より悪くなった「さっきの状態に戻してください。」
一部しか返ってこない「コード全体を出してください。」
音が鳴らない画面を一度クリック。PCの音量とミュートも確認
どうにもならない手を挙げる。それが一番早いです

B. 自分の好きな音楽を使う

手順を見る(5分ほどかかります)

音楽ファイルを置くために、p5.js のアカウントが必要です。

  1. 右上の Sign up で登録(Googleアカウントがあると早い)
  2. 左のファイル一覧「▼」→ Upload file で音楽を追加

アカウントを作ると作品が保存されます。その場合、共有ドキュメントには コードではなくスケッチのURLを貼ってください。
