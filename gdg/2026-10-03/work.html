// おしゃべりボット / ブラウザだけで動きます
//
// --- 変更履歴 ------------------------------------
// v1  決まった言葉に、決まった返事をするだけのボット
// --- ここまで ------------------------------------
// ※ 変更したら「ここまで」の行の1つ上に、v2, v3 … と1行だけ足してください。
//   すでにある行は書き換えないでください。

// 「この言葉が入っていたら、こう返す」を並べたもの
const RULES = [
  { kotoba: 'こんにちは', henji: 'こんにちは。' },
  { kotoba: '名前',       henji: 'まだ名前はありません。' },
  { kotoba: 'ありがとう', henji: 'どういたしまして。' },
];

// どのルールにも当てはまらなかったときの返事
const WAKARANAI = 'よくわかりません。';

// 言われた言葉から、返事を決める
function kangaeru(hatsugen) {
  for (const rule of RULES) {
    if (hatsugen.includes(rule.kotoba)) {
      return rule.henji;
    }
  }
  return WAKARANAI;
}

// ------------------------------------------------------------
// ここから下は、画面に出すための部分です。
// はじめのうちは、さわらなくて大丈夫です。
// ------------------------------------------------------------

const log = document.getElementById('log');
const input = document.getElementById('input');

function hanasu() {
  const hatsugen = input.value.trim();
  if (hatsugen === '') return;

  kakidasu('あなた', hatsugen);
  kakidasu('ボット', kangaeru(hatsugen));

  input.value = '';
  input.focus();
}

function kakidasu(dare, kotoba) {
  const gyou = document.createElement('p');
  gyou.textContent = dare + '：' + kotoba;
  log.appendChild(gyou);
  log.scrollTop = log.scrollHeight;
}

document.getElementById('send').addEventListener('click', hanasu);
input.addEventListener('keydown', function (e) {
  if (e.key === 'Enter') hanasu();
});

3育て方の例

ここから先は、自分の言葉でAIに頼みます。やりたいことを日本語で書くだけです。

書き方の見本 ・「〜してください」と書く
・ひとつだけ書く(まとめて頼むと失敗します)
・最後に「変更したコード全体を出してください」とつけると確実
例: 返事を足す「元気?」と聞かれたら「元気です。」と返すルールを足してください。

やってみたいこと(自分の言葉で頼んでみよう)

見た目を変える

慣れてきたら

聞き返してくるようにする
前に話したことを覚えている

4うまくいかないとき

動いたら保存コードをどこかにコピー
エラーはAIに貼る英語は読まなくていい

下の文はそのまま使ってください。ここは考えなくて大丈夫です。

赤いエラーが出た「次のエラーが出ました。原因を直したコード全体を出してください。」
↑これを送って、そのあとにエラーを全部貼る
送っても返事が出ない「送るボタンを押しても返事が出ません。原因を調べて、動くコード全体を出してください。」
前より悪くなった「さっきの状態に戻してください。」
一部しか返ってこない「コード全体を出してください。」
画面が真っ白貼り忘れたファイルがないか、3つとも確認
どうにもならない手を挙げる。それが一番早いです

5よその席へ行ってみる

  1. 席を立って、ほかの人のボットに話しかけに行く
  2. 打ちこむのは、そのボットを作った人。あなたは声で「こう聞いてみて」と伝える
  3. 崩れたら、それが発見

こんなことを試すと、よく崩れます

言い方を変える「〜は?」→「〜ってどうすればいいんだっけ」
タメ口で聞くていねいな言葉しか想定していないことが多い
長い文で聞く「あのー、ちょっと聞きたいんですけど…」
関係ない話をするまったく別の話題を振ってみる
同じことを2回聞く同じ返事が返ってくるか

席に戻ったら、見つかった穴をふさいでみてください。
ふさぎきれないと気づいたら、それも大事な結果です。
