create table students (
    id varchar(3) primary key,
    name varchar(20) not null
) engine = InnoDB;

create table courses (
    id varchar(3) primary key,
    name varchar(30) not null
) engine = InnoDB;

create table enrollments (
    sid varchar(3),
    cid varchar(3),
    grade varchar(1),
    primary key (sid, cid),
    foreign key (sid)
        references students (id),
    foreign key (cid)
        references courses (id)
) engine = InnoDB;
